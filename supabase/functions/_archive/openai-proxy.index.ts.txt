//  وسيط OpenAI ثانٍ. أفضل حالاً من `OpenAI` (يتحقّق من جلسة مستخدم حقيقية
//  وله حدّ يومي)، لكنه يقبل **أي مستخدم مسجَّل** بلا فحص دور ولا ملكية،
//  ويمرّر الجسم كاملاً — أي أن أي حساب يشغّل أي مطالبة على فاتورة المالك.
//
//  ⚠️ كان عاجزاً **بالصدفة لا بالتصميم**: جداوله الثلاثة `platform_secrets`
//  و`ai_usage` و`founder_config` غير موجودة في هذا المشروع، فيردّ 503. صفٌّ
//  واحد يُدرَج ⇒ يصير حيّاً.
//
//  تخصّ منصّة HSE لا كلاود منيو.
// ════════════════════════════════════════════════════════════════════
import "jsr:@supabase/functions-js/edge-runtime.d.ts";
import { createClient } from "https://esm.sh/@supabase/supabase-js@2";

const cors = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
};

Deno.serve(async (req: Request) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: cors });

  const auth = req.headers.get("Authorization");
  if (!auth) {
    return new Response(JSON.stringify({ error: "Unauthorized" }), { status: 401, headers: cors });
  }

  const SUPABASE_URL = Deno.env.get("SUPABASE_URL")!;
  const SERVICE_KEY = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!;
  const svc = createClient(SUPABASE_URL, SERVICE_KEY);

  const { data: { user }, error } = await svc.auth.getUser(auth.replace("Bearer ", ""));
  if (error || !user) {
    return new Response(JSON.stringify({ error: "Invalid token" }), { status: 401, headers: cors });
  }

  const { data: secretData } = await svc.from("platform_secrets").select("value").eq("key", "openai_api_key").single();
  const OPENAI_KEY = (secretData?.value ?? "").trim();
  if (!OPENAI_KEY) {
    return new Response(JSON.stringify({ error: "لم يتم تكوين مفتاح AI" }), { status: 503, headers: cors });
  }

  const today = new Date().toISOString().split("T")[0];
  const { data: usage } = await svc.from("ai_usage").select("count").eq("user_id", user.id).eq("date", today).maybeSingle();
  const { data: cfg } = await svc.from("founder_config").select("config").eq("id", "main").single();
  const limit: number = cfg?.config?.settings?.ailimit ?? 40;
  const count: number = usage?.count ?? 0;

  if (count >= limit) {
    return new Response(
      JSON.stringify({ error: `وصلت للحد اليومي (${limit} طلب). جرّب غداً.` }),
      { status: 429, headers: cors }
    );
  }

  await svc.from("ai_usage").upsert({ user_id: user.id, date: today, count: count + 1, updated_at: new Date().toISOString() });

  const body = await req.json();
  let oaRes: Response;
  try {
    oaRes = await fetch("https://api.openai.com/v1/chat/completions", {
      method: "POST",
      headers: { "Authorization": `Bearer ${OPENAI_KEY}`, "Content-Type": "application/json" },
      body: JSON.stringify(body),
    });
  } catch (_e) {
    return new Response(JSON.stringify({ error: "تعذّر الاتصال بـ OpenAI" }), { status: 502, headers: cors });
  }

  const data = await oaRes.json();
  return new Response(JSON.stringify(data), {
    status: oaRes.status,
    headers: { ...cors, "Content-Type": "application/json" },
  });
});
