// ════════════════════════════════════════════════════════════
//  Edge Function: ai-proxy
//  الوظيفة: وسيط آمن بين موقعك و OpenAI
//  يشغّل: فريق العمل (لوحة المؤسس) + استيراد المنيو بالصورة
//          + أوصاف الأطباق الذكية + مساعد الدعم
//
//  المسار في مشروعك:  supabase/functions/ai-proxy/index.ts

import { serve } from "https://deno.land/std@0.224.0/http/server.ts";

const CORS = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
};

serve(async (req) => {
  // معالجة preflight
  if (req.method === "OPTIONS") {
    return new Response("ok", { headers: CORS });
  }

  try {
    const OPENAI_API_KEY = Deno.env.get("OPENAI_API_KEY");
    if (!OPENAI_API_KEY) {
      return json({ error: "OPENAI_API_KEY غير مضاف في Supabase Secrets" }, 500);
    }

    const body = await req.json();
    const {
      system = "",
      messages = [],
      model = "gpt-4o-mini",
      max_tokens = 4000,
      temperature = 0.7,
    } = body;

    // بناء رسائل OpenAI
    const oaMessages = [];
    if (system) oaMessages.push({ role: "system", content: system });
    for (const m of messages) {
      oaMessages.push({ role: m.role || "user", content: m.content });
    }

    const resp = await fetch("https://api.openai.com/v1/chat/completions", {
      method: "POST",
      headers: {
        "Authorization": `Bearer ${OPENAI_API_KEY}`,
        "Content-Type": "application/json",
      },
      body: JSON.stringify({
        model,
        messages: oaMessages,
        max_tokens: Math.min(max_tokens || 4000, 8000),
        temperature,
      }),
    });

    const data = await resp.json();

    if (!resp.ok) {
      return json({ error: data?.error?.message || `OpenAI error ${resp.status}` }, resp.status);
    }

    const choice = data.choices?.[0];
    const text = choice?.message?.content || "";
    const finish = choice?.finish_reason;

    // تنبيه لو انقطع الرد بسبب الحد
    if (finish === "length") {
      return json({ text, content: text, truncated: true, note: "الرد طويل وتم اقتطاعه" });
    }

    return json({ text, content: text });
  } catch (e) {
    return json({ error: String(e?.message || e) }, 500);
  }
});

function json(obj: unknown, status = 200) {
  return new Response(JSON.stringify(obj), {
    status,
    headers: { ...CORS, "Content-Type": "application/json" },
  });
}
