//  أداة نقل لمرّة واحدة كُتبت أثناء نقل المشروع (٢٢/٠٨): نقلت ستّ صور من
//  تخزين المشروع القديم ومن أعمدة `data:` base64 إلى دلاء المشروع الجديد.
//  أدّت عملها ورمزها في `internal_secrets` حُذف بعدها، فصارت عاجزة.
//
//  ⚠️ أُقبرت رغم عجزها: دالّة تحمل **مفتاح الخدمة** وتكتب في التخزين وفي
//  ثلاثة جداول لا تُترك منشورة بعد انتهاء مهمّتها. صفٌّ واحد يُدرَج في
//  `internal_secrets` يعيدها حيّة.
//
//  إن لزم نقلٌ مشابه: أعِد نشرها من هذا الملفّ، وولّد رمزاً جديداً،
//  ثم أقبرها مجدداً واحذف الرمز.
// ════════════════════════════════════════════════════════════════════
/**
 * `migrate-images` — أداة نقل لمرّة واحدة، لا ميزة منتج.
 *
 * ═══ لماذا وُجدت ═══
 *
 * صور المطاعم كانت في مكانين خطأ:
 *   ١. `data:image/...;base64,…` **داخل أعمدة نصّية** — تُقرأ في ردّ JSON
 *      حاجز لصفحة المنيو، فيُنزَّل نصف ميغابايت قبل أن تُرسم الصفحة.
 *   ٢. روابط تشير إلى تخزين **المشروع القديم** بعد نقل القاعدة — تعمل ما دام
 *      القديم حيّاً، وتنكسر يوم يُوقَف.
 *
 * فتُقرأ البايتات (فكّ base64 أو تنزيل من الرابط القديم)، وتُرفع إلى دلاء
 * هذا المشروع، ولا يُكتب الرابط الجديد **إلا بعد التحقّق منه فعلاً**: طلب
 * GET على الرابط العام ومقارنة الحجم. صورة مفقودة في منيو حيّ أسوأ من صورة
 * ثقيلة، فالفشل يُبقي القيمة القديمة كما هي.
 *
 * ═══ الحراسة ═══
 *
 * لا جلسة هنا (تُنادى من أداة لا من متصفّح)، فالبوّابة رمزٌ لمرّة واحدة في
 * `internal_secrets.migrate_images_token` — المحجوب عن `anon` و`authenticated`
 * معاً. **احذف الصفّ بعد الانتهاء** فتصير الدالة عاجزة عن فعل شيء.
 */
const URL_ = Deno.env.get("SUPABASE_URL")!;
const SERVICE = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!;

const H = {
  apikey: SERVICE,
  authorization: `Bearer ${SERVICE}`,
  "content-type": "application/json",
};
const json = (b: unknown, s = 200) =>
  new Response(JSON.stringify(b, null, 2), { status: s, headers: { "content-type": "application/json" } });

const EXT: Record<string, string> = {
  "image/jpeg": "jpg", "image/png": "png", "image/webp": "webp",
};

type Job = {
  table: string; column: string; bucket: string; id: string;
  restaurant: string; kind: string; value: string;
};

/** بايتات + نوع، من data URI أو من رابط قديم. */
async function bytesOf(value: string): Promise<{ buf: Uint8Array; mime: string; path?: string } | null> {
  if (value.startsWith("data:")) {
    const m = value.match(/^data:([^;,]+);base64,(.*)$/s);
    if (!m) return null;
    const bin = atob(m[2]);
    const buf = new Uint8Array(bin.length);
    for (let i = 0; i < bin.length; i++) buf[i] = bin.charCodeAt(i);
    return { buf, mime: m[1] };
  }
  // رابط قديم: نحتفظ **بنفس المسار** فلا يتغيّر إلا المضيف.
  const r = await fetch(value);
  if (!r.ok) return null;
  const buf = new Uint8Array(await r.arrayBuffer());
  const mime = r.headers.get("content-type")?.split(";")[0] ?? "image/jpeg";
  const after = value.split("/object/public/")[1] ?? "";
  const path = after.split("/").slice(1).join("/") || undefined;
  return { buf, mime, path };
}

Deno.serve(async (req) => {
  if (req.method !== "POST") return json({ error: "POST" }, 405);

  const token = req.headers.get("x-migrate-token") ?? "";
  const sec = await fetch(
    `${URL_}/rest/v1/internal_secrets?key=eq.migrate_images_token&select=value`,
    { headers: H },
  ).then((r) => r.json());
  const expected = Array.isArray(sec) ? sec[0]?.value : null;
  if (!expected || token !== expected) return json({ error: "forbidden" }, 403);

  const dry = new URL(req.url).searchParams.get("apply") !== "1";

  // ── ما يحتاج نقلاً ───────────────────────────────────────────────────
  const stale = `wjqpsbpebpntpeinqccl.supabase.co`;
  const jobs: Job[] = [];
  const q = async (p: string) => await fetch(`${URL_}/rest/v1/${p}`, { headers: H }).then((r) => r.json());

  for (const r of await q("restaurants?select=id,logo_image,banner_image")) {
    for (const [col, kind] of [["logo_image", "logo"], ["banner_image", "banner"]] as const) {
      const v = r[col];
      if (v && (v.startsWith("data:") || v.includes(stale))) {
        jobs.push({ table: "restaurants", column: col, bucket: "restaurant-images", id: r.id, restaurant: r.id, kind, value: v });
      }
    }
  }
  for (const d of await q("dishes?select=id,restaurant_id,image")) {
    if (d.image && (d.image.startsWith("data:") || d.image.includes(stale))) {
      jobs.push({ table: "dishes", column: "image", bucket: "dish-images", id: d.id, restaurant: d.restaurant_id, kind: "dish", value: d.image });
    }
  }
  for (const m of await q("menus?select=id,restaurant_id,cover_image")) {
    if (m.cover_image && (m.cover_image.startsWith("data:") || m.cover_image.includes(stale))) {
      jobs.push({ table: "menus", column: "cover_image", bucket: "menu-images", id: m.id, restaurant: m.restaurant_id, kind: "cover", value: m.cover_image });
    }
  }

  const report: Record<string, unknown>[] = [];
  if (dry) {
    return json({ dry_run: true, count: jobs.length, jobs: jobs.map((j) => ({ ...j, value: j.value.slice(0, 60), bytes: j.value.length })) });
  }

  for (const j of jobs) {
    const got = await bytesOf(j.value);
    if (!got) { report.push({ ...j, value: undefined, ok: false, why: "unreadable" }); continue; }

    const ext = EXT[got.mime] ?? "jpg";
    const path = got.path ??
      `${j.restaurant}/${j.kind}/${Date.now()}-${Math.random().toString(36).slice(2, 8)}.${ext}`;

    const up = await fetch(`${URL_}/storage/v1/object/${j.bucket}/${path}`, {
      method: "POST",
      headers: { apikey: SERVICE, authorization: `Bearer ${SERVICE}`, "content-type": got.mime, "x-upsert": "true" },
      body: got.buf,
    });
    if (!up.ok) { report.push({ table: j.table, id: j.id, ok: false, why: `upload ${up.status} ${(await up.text()).slice(0, 120)}` }); continue; }

    // ⚠️ التحقّق قبل الكتابة: الرابط يُطلب فعلاً ويُقارن حجمه.
    const publicUrl = `${URL_}/storage/v1/object/public/${j.bucket}/${path}`;
    const check = await fetch(publicUrl);
    const size = check.ok ? (await check.arrayBuffer()).byteLength : -1;
    if (size !== got.buf.length) {
      report.push({ table: j.table, id: j.id, ok: false, why: `verify ${check.status} size=${size} want=${got.buf.length}` });
      continue;
    }

    const patch = await fetch(`${URL_}/rest/v1/${j.table}?id=eq.${j.id}`, {
      method: "PATCH",
      headers: { ...H, Prefer: "return=minimal" },
      body: JSON.stringify({ [j.column]: publicUrl }),
    });
    report.push({ table: j.table, column: j.column, id: j.id, ok: patch.ok, bytes: got.buf.length, path, status: patch.status });
  }

  return json({ applied: true, count: jobs.length, report });
});
